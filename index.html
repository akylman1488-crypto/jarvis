#include <Servo.h>
Servo jarvisServo;
const int PIN_RELAY = 7, PIN_SERVO = 9, PIN_TRIG = 10, PIN_ECHO = 11;
String cmd = "";

long readRadarCm() {
  digitalWrite(PIN_TRIG, LOW); delayMicroseconds(2);
  digitalWrite(PIN_TRIG, HIGH); delayMicroseconds(10);
  digitalWrite(PIN_TRIG, LOW);
  long dur = pulseIn(PIN_ECHO, HIGH, 25000);
  return dur == 0 ? 120 : dur * 0.034 / 2;
}

void setup() {
  Serial.begin(115200);
  pinMode(PIN_RELAY, OUTPUT); pinMode(PIN_TRIG, OUTPUT); pinMode(PIN_ECHO, INPUT);
  jarvisServo.attach(PIN_SERVO); jarvisServo.write(90);
}

void loop() {
  while (Serial.available() > 0) {
    char c = Serial.read();
    if (c == '\n') {
      cmd.trim();
      if (cmd == "RELAY:1") digitalWrite(PIN_RELAY, HIGH);
      else if (cmd == "RELAY:0") digitalWrite(PIN_RELAY, LOW);
      else if (cmd.startsWith("SERVO:")) jarvisServo.write(constrain(cmd.substring(6).toInt(), 0, 180));
      Serial.print("{\"cmd\":\""); Serial.print(cmd);
      Serial.print("\",\"cm\":"); Serial.print(readRadarCm()); Serial.println("}");
      cmd = "";
    } else cmd += c;
  }
}
